# Third-party tools that CI installs by name, pinned to an exact version so a
# run's result depends on this commit and not on what PyPI served that morning.
# Dependabot watches this file (the "/ci" pip entry in .github/dependabot.yml),
# so an upgrade arrives as a PR someone reads.
#
# Only the gate's own tools belong here. The project's toolchain (ruff, pytest,
# pyright) stays in pyproject.toml's dev extras with a version bound, and CI
# installs it with `uv pip install -e ".[dev]"`.
#
# uv is the installer everywhere -- CI, the devcontainer and the Dockerfile --
# and the build frontend (`uv build`, which replaced `python -m build`, so
# `build` left this file). Its pin lives on the line below and nowhere else:
# every setup-uv step reads it via `version-file: ci/requirements.txt` and
# setup.sh greps it, so one Dependabot PR moves every environment at once.

uv==0.12.5  # installer, resolver, venv manager and build frontend
